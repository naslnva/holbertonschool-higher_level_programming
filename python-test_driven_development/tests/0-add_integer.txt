add_integer = __import__('0-add_integer').add_integer

# Regular tests
add_integer(1, 2)
# 3
add_integer(100, -2)
# 98
add_integer(2)
# 100
add_integer(100.3, -2)
# 98

# TypeError tests
try:
    add_integer(4, "School")
except Exception as e:
    print(e)
# b must be an integer

try:
    add_integer(None)
except Exception as e:
    print(e)
# a must be an integer

# More tests
add_integer(-1, -1)
# -2
add_integer(0, 0)
# 0

# NaN test
import math
try:
    add_integer(math.nan, 1)
except Exception as e:
    print(e)
# cannot convert float NaN to integer

# Infinity test
try:
    add_integer(float("inf"), 1)
except Exception as e:
    print(e)
# cannot convert float infinity to integer
